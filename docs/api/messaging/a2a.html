messaging.a2a

A2A 1.0 protocol objects, builders, validators and errors, using only the stdlib.

Field names follow the A2A 1.0 JSON (camelCase); there is no kind member: a Part is told apart by which of text/data/raw/url it holds. Enums use their full names (TASK_STATE_COMPLETED, ROLE_USER) and timestamps are ISO-8601 UTC with millisecond precision and a Z suffix. See specs/002-qa-feedback-loop-mcp/contracts/a2a-binding.md.

A2A_VERSION = '1.0'
BINDING_URN = 'urn:multi-agent-lab:a2a-binding:asyncio-broker:1.0'
TaskState = typing.Literal['TASK_STATE_SUBMITTED', 'TASK_STATE_WORKING', 'TASK_STATE_COMPLETED', 'TASK_STATE_FAILED', 'TASK_STATE_CANCELED', 'TASK_STATE_REJECTED', 'TASK_STATE_INPUT_REQUIRED', 'TASK_STATE_AUTH_REQUIRED']
Role = typing.Literal['ROLE_USER', 'ROLE_AGENT']
ROLES = ('ROLE_USER', 'ROLE_AGENT')
TERMINAL_STATES = frozenset({'TASK_STATE_FAILED', 'TASK_STATE_COMPLETED', 'TASK_STATE_CANCELED', 'TASK_STATE_REJECTED'})
PART_CONTENT_KEYS = ('text', 'data', 'raw', 'url')
SUPPORTED_OUTPUT_MODES = frozenset({'application/json', 'text/plain'})
ERROR_CODES = {'TaskNotFoundError': -32001, 'TaskNotCancelableError': -32002, 'PushNotificationNotSupportedError': -32003, 'UnsupportedOperationError': -32004, 'ContentTypeNotSupportedError': -32005, 'InvalidAgentResponseError': -32006, 'ExtendedAgentCardNotConfiguredError': -32007, 'ExtensionSupportRequiredError': -32008, 'VersionNotSupportedError': -32009, 'MethodNotFound': -32601, 'InvalidParams': -32602, 'UnknownAgent': -32602, 'InternalError': -32603, 'BrokerClosedError': -32603, 'A2ATimeoutError': -32603}
